cmake_minimum_required(VERSION 3.20)
project(omarchy_trofeo VERSION 0.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
add_executable(trofeo-engine src/engine.cpp src/themes.cpp)
qt_add_resources(trofeo-engine "translations" PREFIX "/" FILES i18n/messages.json)
target_link_libraries(trofeo-engine PRIVATE Qt6::Core Qt6::Gui Qt6::Network PkgConfig::LIBUSB)
target_compile_options(trofeo-engine PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME protocol-and-render COMMAND trofeo-engine --self-test)
add_executable(themes-test tests/themes_test.cpp src/themes.cpp)
qt_add_resources(themes-test "translations" PREFIX "/" FILES i18n/messages.json)
target_include_directories(themes-test PRIVATE src)
target_link_libraries(themes-test PRIVATE Qt6::Core Qt6::Gui)
target_compile_options(themes-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME custom-theme-library COMMAND themes-test)
set_tests_properties(protocol-and-render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
